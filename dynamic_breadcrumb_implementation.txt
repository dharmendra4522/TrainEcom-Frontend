HOW TO IMPLEMENT DYNAMIC BREADCRUMBS IN ANGULAR
===============================================

Follow these steps to create and integrate a dynamic breadcrumb component in your project.

STEP 1: Update Route Configurations
-----------------------------------
Add `data: { breadcrumb: '...' }` to the routes you want to display in the breadcrumb path.

File: src/app/app-routing.module.ts
-----------------------------------
Modify your route definition like this:

const routes: Routes = [
  {
    path: '',
    component: MainLayoutComponent,
    children: [
      {
        path: '',
        loadChildren: () => import('./features/home/home.module').then((m) => m.HomeModule),
      },
      {
        path: 'account',
        canActivate: [authGuard],
        data: { breadcrumb: 'My Account' }, // <-- Add this parent label
        loadChildren: () => import('./features/users/users.module').then((m) => m.UsersModule),
      },
      {
        path: 'admin',
        canActivate: [authGuard, adminGuard],
        data: { breadcrumb: 'Admin' }, // <-- Add this parent label
        children: [
          {
            path: 'dashboard',
            data: { breadcrumb: 'Dashboard' }, // <-- Add this
            loadChildren: () => import('./features/dashboard/dashboard.module').then((m) => m.DashboardModule),
          },
          {
            path: 'products',
            data: { breadcrumb: 'Products' }, // <-- Add this
            loadChildren: () => import('./features/products/products.module').then((m) => m.ProductsModule),
          },
          ...
        ],
      },
    ],
  },
];

STEP 2: Create the Breadcrumb Component
---------------------------------------
Create a new reusable breadcrumb component in your shared components directory.

File: src/app/shared/components/breadcrumb/breadcrumb.component.ts
-------------------------------------------------------------------
import { Component, OnInit } from '@angular/core';
import { ActivatedRoute, NavigationEnd, Router } from '@angular/router';
import { filter } from 'rxjs/operators';

interface Breadcrumb {
  label: string;
  url: string;
}

@Component({
  selector: 'app-breadcrumb',
  templateUrl: './breadcrumb.component.html',
  styleUrls: ['./breadcrumb.component.css']
})
export class BreadcrumbComponent implements OnInit {
  breadcrumbs: Breadcrumb[] = [];

  constructor(private router: Router, private route: ActivatedRoute) {}

  ngOnInit(): void {
    // Generate breadcrumbs on initial load
    this.breadcrumbs = this.buildBreadcrumbs(this.route.root);

    // Update breadcrumbs on every route change
    this.router.events.pipe(
      filter(event => event instanceof NavigationEnd)
    ).subscribe(() => {
      this.breadcrumbs = this.buildBreadcrumbs(this.route.root);
    });
  }

  private buildBreadcrumbs(route: ActivatedRoute, url: string = '', breadcrumbs: Breadcrumb[] = []): Breadcrumb[] {
    const children: ActivatedRoute[] = route.children;

    if (children.length === 0) {
      return breadcrumbs;
    }

    for (const child of children) {
      const routeURL: string = child.snapshot.url.map(segment => segment.path).join('/');
      if (routeURL !== '') {
        url += `/${routeURL}`;
      }

      const label = child.snapshot.data['breadcrumb'];
      if (label) {
        breadcrumbs.push({ label, url });
      }

      return this.buildBreadcrumbs(child, url, breadcrumbs);
    }

    return breadcrumbs;
  }
}

File: src/app/shared/components/breadcrumb/breadcrumb.component.html
---------------------------------------------------------------------
<nav class="breadcrumb-nav" *ngIf="breadcrumbs.length > 0">
  <a routerLink="/">Home</a>
  <span *ngFor="let breadcrumb of breadcrumbs; let last = last">
    <span class="separator">/</span>
    <a *ngIf="!last" [routerLink]="breadcrumb.url">{{ breadcrumb.label }}</a>
    <span *ngIf="last" class="active">{{ breadcrumb.label }}</span>
  </span>
</nav>

STEP 3: Register in Shared Module
---------------------------------
File: src/app/shared/shared.module.ts
-------------------------------------
Declare and export the new `BreadcrumbComponent` in your SharedModule:

import { BreadcrumbComponent } from './components/breadcrumb/breadcrumb.component';

@NgModule({
  declarations: [
    ...,
    BreadcrumbComponent
  ],
  imports: [
    CommonModule,
    RouterModule,
    ...
  ],
  exports: [
    ...,
    BreadcrumbComponent
  ]
})
export class SharedModule { }


STEP 4: Include in Layout or Pages
----------------------------------
Now you can remove hardcoded breadcrumbs from individual files (like `profile.component.html`) and place the shared component in a global layout, or just use `<app-breadcrumb></app-breadcrumb>` inside any component template:

File: src/app/features/users/components/profile/profile.component.html
----------------------------------------------------------------------
Replace:
<div class="breadcrumb" *appRole="'Customer'">
    <a routerLink="/">Home</a> / <span>My Account</span>
</div>

With:
<app-breadcrumb></app-breadcrumb>
